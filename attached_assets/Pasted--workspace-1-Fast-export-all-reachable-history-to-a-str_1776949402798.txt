~/workspace$ # 1. Fast-export all reachable history to a stream file
cd /home/runner/workspace
git fast-export --all --signed-tags=strip > /tmp/ga.fastexport 2> /tmp/ga.fastexport.err
echo "fast-export exit: $?  (0 = success)"
echo "stream size: $(wc -c < /tmp/ga.fastexport) bytes, $(wc -l < /tmp/ga.fastexport) lines"
echo "---any errors?---"
head -5 /tmp/ga.fastexport.err 2>/dev/null

# 2. Build a fresh clean repo from that stream
rm -rf /tmp/platform-clean
git init -b main /tmp/platform-clean
cd /tmp/platform-clean
git fast-import < /tmp/ga.fastexport 2>&1 | tail -5

# 3. Sanity check
echo "---log of new repo---"
git log --oneline -5
echo "total commits: $(git rev-list --count main)"
echo "---fsck on new repo---"
git fsck --full 2>&1 | grep -iE "missing|broken|error" | head -5 || echo "clean ✓"

# 4. Push from the clean repo to platform
git remote add origin https://github.com/GeneratedArt/platform.git
git -c credential.helper='!f() { echo username=x-access-token; echo "password=$GITHUB_PAT"; }; f' \
    push -u origin main 2>&1 | tail -8

# 5. Verify
echo "---verify---"
curl -s -H "Authorization: Bearer $GITHUB_PAT" https://api.github.com/repos/GeneratedArt/platform/commits/main \
  | node -e 'let d="";process.stdin.on("data",c=>d+=c).on("end",()=>{const j=JSON.parse(d);console.log(j.sha?"remote HEAD: "+j.sha.slice(0,10)+" - "+j.commit.message.split("\n")[0]:"ERROR: "+(j.message||""))})'
